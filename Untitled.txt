    document.documentElement.setAttribute(
      "data-theme", next
    );

    try {
      localStorage.setItem("cxp_theme", next);
    } catch (_) {}

  });

  /* TELEGRAM MODAL */

  function dismissModal() {
    modal.classList.add("hidden");

    try {
      sessionStorage.setItem("cxp_joined", "1");
    } catch (_) {}
  }

  document.getElementById("joinButton")
    .addEventListener("click", function () {

      window.open(
        TELEGRAM_URL,
        "_blank",
        "noopener,noreferrer"
      );

      dismissModal();

    });

  document.getElementById("closeButton")
    .addEventListener("click", dismissModal);

  modal.addEventListener("click", function (e) {
    if (e.target === modal) {
      dismissModal();
    }
  });

  document.addEventListener("keydown", function (e) {

    if (
      e.key === "Escape" &&
      !modal.classList.contains("hidden")
    ) {
      dismissModal();
    }

  });

  try {
    if (sessionStorage.getItem("cxp_joined") === "1") {
      modal.classList.add("hidden");
    }
  } catch (_) {}

  /* TOAST */

  function showToast(message) {

    toast.textContent = message;
    toast.classList.add("show");

    clearTimeout(toastTimer);

    toastTimer = setTimeout(function () {
      toast.classList.remove("show");
    }, 2500);

  }

  /* LOADER */

  function showLoading() {

    loading.classList.remove("hidden");
    loading.style.display = "flex";

    loading.innerHTML = `
      <div class="spinner"></div>
      <span class="loading-text">
        Loading stream...
      </span>
    `;

  }

  function hideLoading() {

    loading.classList.add("hidden");

    setTimeout(function () {
      if (loading.classList.contains("hidden")) {
        loading.style.display = "none";
      }
    }, 300);

  }

  /* ERROR */

  function showError(message) {

    loading.classList.remove("hidden");
    loading.style.display = "flex";
    loading.innerHTML = "";

    const box = document.createElement("div");
    box.className = "error-box";

    const heading = document.createElement("b");
    heading.textContent = "Unable to load stream";

    const description = document.createElement("small");
    description.textContent = message;

    const retryBtn = document.createElement("button");
    retryBtn.textContent = "Retry";

    retryBtn.addEventListener("click", function () {

      if (currentServer) {
        loadStaticServer(currentServer);
      } else {
        initPlayer();
      }

    });

    box.append(heading, description, retryBtn);
    loading.appendChild(box);

  }

  /* CREATE IFRAME */

  function createPlayerIframe(url, title, attempt) {

    const iframe = document.createElement("iframe");

    iframe.src = url;

    iframe.title =
      title || "RUNMACHINE Live Stream";

    iframe.setAttribute(
      "allow",
      "autoplay; encrypted-media; fullscreen; picture-in-picture"
    );

    iframe.setAttribute("allowfullscreen", "");
    iframe.setAttribute("webkitallowfullscreen", "");
    iframe.setAttribute("loading", "eager");

    iframe.style.width = "100%";
    iframe.style.height = "100%";
    iframe.style.border = "none";
    iframe.style.display = "block";

    iframe.addEventListener("load", function () {

      if (attempt === loadAttempt) {
        hideLoading();
      }

    });

    iframe.addEventListener("error", function () {

      if (attempt === loadAttempt) {
        showError(
          "Unable to load the embedded player."
        );
      }

    });

    playerContainer.appendChild(iframe);

  }

  /* ORIGINAL JSON ID PLAYER */

  async function initPlayer() {

    const attempt = ++loadAttempt;

    currentServer = null;
    updateActiveServer(null);

    showLoading();
    playerContainer.replaceChildren();

    if (!id) {
      showError(
        "Select a streaming server below to watch."
      );
      return;
    }

    try {

      const response = await fetch(API_URL);

      if (!response.ok) {
        throw new Error(
          "API request failed: HTTP " + response.status
        );
      }

      const data = await response.json();

      if (attempt !== loadAttempt) return;

      if (
        !data.iframes ||
        !Array.isArray(data.iframes)
      ) {
        throw new Error(
          "Invalid iframe data received from API."
        );
      }

      const match = data.iframes.find(function (item) {
        return String(item.id) === String(id);
      });

      if (!match || !match.iframeSrc) {
        throw new Error(
          "Invalid or unavailable stream ID."
        );
      }

      const iframeUrl = new URL(
        match.iframeSrc,
        window.location.href
      );

      if (
        !["https:", "http:"].includes(
          iframeUrl.protocol
        )
      ) {
        throw new Error("Invalid iframe URL.");
      }

      createPlayerIframe(
        iframeUrl.href,
        "RUNMACHINE Live Stream",
        attempt
      );

    } catch (error) {

      if (attempt !== loadAttempt) return;

      console.error(
        "RUNMACHINE Player Error:",
        error
      );

      showError(
        error.message || "Stream could not be loaded."
      );

    }
  }

  /* STATIC SERVER PLAYER */

  function loadStaticServer(server) {

    const attempt = ++loadAttempt;

    currentServer = server;

    showLoading();
    playerContainer.replaceChildren();

    let iframeUrl;

    try {

      iframeUrl = new URL(
        server.url,
        window.location.href
      );

      if (
        !["https:", "http:"].includes(
          iframeUrl.protocol
        )
      ) {
        throw new Error("Invalid iframe URL.");
      }

    } catch (error) {

      showError("Invalid server URL.");
      return;

    }

    updateActiveServer(server.id);

    createPlayerIframe(
      iframeUrl.href,
      server.name + " - RUNMACHINE Live",
      attempt
    );

    console.log(
      "RUNMACHINE server selected:",
      server.name
    );

  }

  /* SERVER ACTIVE STATE */

  function updateActiveServer(serverId) {

    serverGrid.querySelectorAll(".server-btn")
      .forEach(function (button) {

        const isActive =
          button.dataset.server === serverId;

        button.classList.toggle(
          "active",
          isActive
        );

        button.setAttribute(
          "aria-pressed",
          String(isActive)
        );

      });

  }

  /* RENDER SERVER BUTTONS */

  function renderServerButtons() {

    serverGrid.replaceChildren();

    STREAM_SERVERS.forEach(function (server) {

      const button =
        document.createElement("button");

      button.type = "button";
      button.className = "server-btn";
      button.dataset.server = server.id;

      button.setAttribute(
        "aria-pressed",
        "false"
      );

      const dot =
        document.createElement("span");

      dot.className = "server-dot";

      const name =
        document.createElement("span");

      name.textContent = server.name;

      button.append(dot, name);

      button.addEventListener("click", function () {
        loadStaticServer(server);
      });

      serverGrid.appendChild(button);

    });

  }

  /* SHARE */

  async function copyStreamLink(url) {

    if (
      navigator.clipboard &&
      window.isSecureContext
    ) {

      await navigator.clipboard.writeText(url);

    } else {

      const input =
        document.createElement("textarea");

      input.value = url;
      input.style.position = "fixed";
      input.style.opacity = "0";

      document.body.appendChild(input);
      input.select();

      const copied =
        document.execCommand("copy");

      input.remove();

      if (!copied) {
        throw new Error("Copy failed");
      }

    }

    showToast("Link copied to clipboard");

  }

  shareBtn.addEventListener(
    "click",
    async function () {

      const url = window.location.href;

      try {

        if (navigator.share) {

          await navigator.share({
            title: "RUNMACHINE Live",
            text: "Watch this live stream on RUNMACHINE",
            url: url
          });

        } else {

          await copyStreamLink(url);

        }

      } catch (error) {

        if (error.name !== "AbortError") {

          try {
            await copyStreamLink(url);
          } catch (_) {
            showToast("Unable to copy link");
          }

        }

      }

    }
  );

  /* START */

  renderServerButtons();

  if (id) {

    initPlayer();

  } else if (STREAM_SERVERS.length > 0) {

    loadStaticServer(STREAM_SERVERS[0]);

  } else {

    showError(
      "No streaming servers configured."
    );

  }

});
